\subsection{Bayesian Trust Model}
% PROSE: 2 paragraphs introducing the formalism below.

\paragraph{Update rule.}
After agent $i$ observes seat $j$ take action $a$ in round $r$, the
posterior $p_{i \to j}$ updates as
\begin{equation}
  p_{i \to j}(t)\,{\propto}\,
    \bigl[(1{-}\varepsilon)\,L(a\mid t,r)
          + \varepsilon/|\mathcal{T}|\bigr]\,
    p_{i \to j}^{\text{prior}}(t),
\end{equation}
where $L(a\mid t,r)$ is the precomputed likelihood that a type-$t$
agent emits action $a$ in round $r$, and $\varepsilon = 0.05$ is a
noise floor preventing any type from being fully eliminated.

\paragraph{Decay.}
Between hands, posteriors shrink toward the uniform prior:
\begin{equation}
  p_{i \to j} \;\leftarrow\;
    \lambda\,p_{i \to j} + (1{-}\lambda)\,|\mathcal{T}|^{-1},
  \qquad \lambda = 0.95.
\end{equation}

\paragraph{Trust score.}
\begin{equation}
  \text{trust}_{i \to j} \;=\;
    \sum_{t \in \mathcal{T}}\, p_{i \to j}(t)\, h(t),
\end{equation}
where $h(t)$ is the honesty of archetype $t$ shown in Fig.~\ref{fig:archetypes}.

\subsection{Metrics}
% PROSE: 1 paragraph introducing the list.
\begin{itemize}\setlength{\itemsep}{1pt}\setlength{\parskip}{0pt}
  \item \textit{Trust--profit r} -- Pearson correlation between mean
    trust score and final stack across the eight archetypes.
  \item \textit{Trust Exploitation Index (TEI)} -- non-showdown chips
    won per hand played.
  \item \textit{Context Sensitivity (CS)} -- aggression variance
    across hand-strength buckets.
  \item \textit{Opponent Adaptation (OA)} -- std of aggression rate
    across opponents.
  \item \textit{Non-Stationarity (NS)} -- shift in action distribution
    over time.
  \item \textit{Strategic Unpredictability (SU)} -- Shannon entropy
    in bits over the action distribution.
  \item \textit{Trust Manipulation Awareness (TMA)} -- correlation
    between rolling trust delta and rolling aggression delta.
\end{itemize}
