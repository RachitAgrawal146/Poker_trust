\begin{itemize}\setlength{\itemsep}{1pt}
  \item 8 seats, dealer button rotates clockwise per hand.
  \item Small blind $= 1$, big blind $= 2$.
  \item Small bet $= 2$ (preflop/flop), big bet $= 4$ (turn/river).

\subsection{Bayesian Trust Model}
\label{sec:trust-model}
% PROSE: 2 paragraphs.
%   (1) The posterior — length-8 categorical per (observer, observee).
%       Updated on every action; refined at every showdown.
%   (2) The mechanics — likelihood, epsilon noise, decay, trust score.

\paragraph{Update rule.}
After agent $i$ observes seat $j$ take action $a$ in round $r$, the
posterior $p_{i \to j}$ is updated as
\begin{equation}
  p_{i \to j}(t) \;\propto\; \bigl[(1-\varepsilon)\,L(a \mid t, r)
                              + \varepsilon/|\mathcal{T}|\bigr]
                              \,p_{i \to j}^{\text{prior}}(t),
  \label{eq:posterior}
\end{equation}
where $L(a\mid t,r)$ is the precomputed likelihood that a type-$t$
agent emits action $a$ in round $r$, $\mathcal{T}$ is the set of
eight archetypes, and $\varepsilon = 0.05$ is the noise floor.

\paragraph{Decay.}
Between hands, posteriors are shrunk toward the uniform prior:
\begin{equation}
  p_{i \to j} \;\leftarrow\; \lambda\,p_{i \to j}
                             + (1-\lambda)\,|\mathcal{T}|^{-1},
  \qquad \lambda = 0.95.
  \label{eq:decay}
\end{equation}

\paragraph{Trust score.}
Trust is the honesty-weighted expectation under the posterior:
\begin{equation}
  \text{trust}_{i \to j} \;=\; \sum_{t \in \mathcal{T}}\,p_{i \to j}(t)\,h(t),
  \label{eq:trust}
\end{equation}
where $h(t)$ is the archetype's honesty score (Table~\ref{tab:archetypes}).

\subsection{Metrics}
\label{sec:metrics}
% PROSE: 1 paragraph + the list below.
\begin{itemize}\setlength{\itemsep}{1pt}
  \item \textit{Trust--profit~r} -- Pearson correlation between mean
    trust score (final hand) and final stack, across the eight
    archetypes within a seed.
  \item \textit{Trust Exploitation Index (TEI)} -- chips won non-showdown
    per hand played.
  \item \textit{Context Sensitivity (CS)} -- aggression variance across
    hand-strength buckets.
  \item \textit{Opponent Adaptation (OA)} -- per-agent std of aggression
    rate across opponents.
  \item \textit{Non-Stationarity (NS)} -- shift in action distribution
    over time.
  \item \textit{Strategic Unpredictability (SU)} -- Shannon entropy in
    bits over the action distribution.
  \item \textit{Trust Manipulation Awareness (TMA)} -- correlation
    between rolling trust delta and rolling aggression delta.
\end{itemize}

\subsection{Limitations}
\label{sec:limitations}
% PROSE: 1 sentence intro, then the list.

\noindent Six design choices constrain generalizability:
\begin{enumerate}\setlength{\itemsep}{2pt}
  \item \textit{Bounded optimization, not unrestricted learning.}
    Phase 2 adapts within bound boxes; Phase 2$^{*}$ lifts these but
    uses coordinate descent rather than CMA-ES or REINFORCE.
  \item \textit{No per-opponent state in Phases 1--2.}
    The hill-climber has no slot for ``what to do against Wall vs.\
    Firestorm.''
  \item \textit{Limit Hold'em.} Fixed-limit betting constrains
    aggression; no-limit would amplify the trap.
  \item \textit{Eight archetypes.} Hand-selected; a larger or randomly
    sampled population could produce different dynamics.
  \item \textit{Single game environment.} Generalization to auctions,
    negotiations, social dilemmas requires direct validation.
  \item \textit{Trust-model specificity.} A specific Bayesian
    parameterization ($\lambda = 0.95$, $\varepsilon = 0.05$). Limited
    sensitivity sweep suggests robustness; comprehensive sweep is
    future work.
\end{enumerate}
