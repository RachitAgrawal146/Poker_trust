\noindent The game environment is fully fixed across every phase:
\begin{itemize}
  \item 8 seats, dealer button rotates clockwise per hand.
  \item Small blind $= 1$, big blind $= 2$.
  \item Small bet $= 2$ (preflop/flop), big bet $= 4$ (turn/river).
  \item Bet cap of 4 per street (no-raise after the 4th aggression).
  \item Starting stack $= 200$. Rebuy granted when a seat busts (stack $= 0$).
  \item Showdown awarded by side pots per contribution level
    (see Section~\ref{sec:env-sidepot}); ties split, remainder to
    earliest clockwise seat from the dealer.
\end{itemize}

\subsection{Bayesian Trust Model}
\label{sec:trust-model}
% PROSE: 2 paragraphs.
%   (1) The posterior. Every agent maintains a length-8 categorical
%       distribution over archetypes for every other seat. Updated on
%       every observed action (live, bucket unknown) and refined at
%       every showdown when hole cards reveal the actual hand-strength
%       bucket.
%   (2) The mechanics — likelihood with epsilon-noise smoothing, decay,
%       third-party weight. State the parameters used throughout:
%       lambda = 0.95, epsilon = 0.05.

\paragraph{Update rule.}
After agent $i$ observes seat $j$ take action $a$ in round $r$, the
posterior $p_{i \to j}$ over archetype types $t$ is updated as
\begin{equation}
  p_{i \to j}(t) \;\propto\; \big[\,(1-\varepsilon)\,L(a \mid t, r)
                                + \varepsilon \cdot \tfrac{1}{|\mathcal{T}|}\,\big]
                              \cdot p_{i \to j}^{\text{prior}}(t),
  \label{eq:posterior}
\end{equation}
where $L(a \mid t, r)$ is the precomputed likelihood that a type-$t$
agent emits action $a$ in round $r$, $\mathcal{T}$ is the set of eight
archetypes, and $\varepsilon = 0.05$ is the noise floor that prevents
any type from being eliminated.

\paragraph{Decay.}
Between hands, every posterior is shrunk toward the uniform prior by
\begin{equation}
  p_{i \to j} \;\leftarrow\; \lambda\,p_{i \to j} + (1-\lambda)\,\frac{1}{|\mathcal{T}|},
  \qquad \lambda = 0.95.
  \label{eq:decay}
\end{equation}

\paragraph{Trust score.}
Trust is the honesty-weighted expectation under the posterior:
\begin{equation}
  \text{trust}_{i \to j} \;=\; \sum_{t \in \mathcal{T}} p_{i \to j}(t) \cdot h(t),
  \label{eq:trust}
\end{equation}
where $h(t)$ is the archetype's honesty score (Table~\ref{tab:archetypes}).

\subsection{Metrics Framework}
\label{sec:metrics}
% PROSE: 1 paragraph + an itemize. Define the six measured dimensions
% briefly. Cite analysis/compute_metrics.py.
\begin{itemize}
  \item \textbf{Trust--profit r} -- Pearson correlation between mean
    trust score (final hand) and final stack across the eight archetypes.
  \item \textbf{Trust Exploitation Index (TEI)} -- chips won non-showdown
    per hand played; positive means the agent extracts uncalled fold-equity value.
  \item \textbf{Context Sensitivity (CS)} -- variance of aggression rate
    across hand-strength buckets.
  \item \textbf{Opponent Adaptation (OA)} -- per-agent standard deviation
    of aggression rate across opponents.
  \item \textbf{Non-Stationarity (NS)} -- whether the action distribution
    shifts over time within a run.
  \item \textbf{Strategic Unpredictability (SU)} -- Shannon entropy in
    bits over the action distribution.
  \item \textbf{Trust Manipulation Awareness (TMA)} -- correlation
    between rolling trust delta and rolling aggression delta.
\end{itemize}

\subsection{Limitations}
\label{sec:limitations}
% PROSE: a paragraph introducing the list, then the six numbered points
% below. Replace the placeholder narrative with your own opening.

\noindent Six design choices constrain the generalizability of our findings:
\begin{enumerate}
  \item \textbf{Bounded optimization, not unrestricted learning.}
    Phase 2 agents adapt within their archetype bound box; Phase 2$^*$
    lifts this but uses coordinate descent rather than REINFORCE or
    CMA-ES. The $\Delta\rtp = +0.116$ softening is a lower bound on what
    numerical adaptation can produce.
  \item \textbf{No per-opponent state in Phases 1--2.}
    The hill-climber optimizes aggregate profit and has no slot for
    ``what to do against Wall vs.\ Firestorm.'' OA $\approx 0$ is partly
    a consequence of this design.
  \item \textbf{Limit Hold'em.}
    The fixed-limit betting structure constrains aggression. No-limit
    variants would amplify fold equity and likely deepen the trap.
  \item \textbf{Eight archetypes.}
    Hand-selected to span the strategic space from game theory and
    behavioral economics; a larger or randomly generated population
    might produce different dynamics.
  \item \textbf{Single game environment.}
    Findings are specific to poker. Generalization to other repeated
    strategic interactions with reputation systems (auctions,
    negotiations, social dilemmas) requires direct validation.
  \item \textbf{Trust model specificity.}
    A specific Bayesian parameterization ($\lambda = 0.95$,
    $\varepsilon = 0.05$). Limited sensitivity analysis suggests
    robustness across $\lambda \in [0.90, 1.00]$, but a comprehensive
    sweep is left to future work.
\end{enumerate}
